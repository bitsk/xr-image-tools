function(export_header name directive)
  set(MY_EXPORT_HEADER_PREFIX "${name}")
  set(MY_LIBNAME "${name}")
  set(MY_LIBRARY_EXPORT_DIRECTIVE "${directive}")
  if(name STREQUAL "vtkMRML")
    set(Slicer_EXPORT_HEADER_CUSTOM_CONTENT "#include <vtkMRMLConfigure.h>")
  endif()
  configure_file("${SLICER}/CMake/qSlicerExport.h.in" "${generated}/${name}Export.h")
endfunction()

function(slicer_library name directory)
  file(GLOB sources "${EXTENSION}/VirtualReality/${directory}/*.cxx")
  list(FILTER sources EXCLUDE REGEX "OpenVR")
  add_library(${name} SHARED ${sources} ${ARGN})
  target_link_libraries(${name} PUBLIC Qt5::Core Qt5::Gui Qt5::Widgets Qt5::OpenGL
    ${VTK_LIBRARIES} VTK::RenderingVR VTK::RenderingOpenXR ${runtime_libraries})
  target_link_options(${name} PRIVATE -Wl,--no-undefined)
  set_target_properties(${name} PROPERTIES AUTOMOC ON AUTOUIC ON AUTORCC ON
    AUTOUIC_SEARCH_PATHS "${EXTENSION}/VirtualReality/${directory}/Resources/UI"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/Slicer-5.11/qt-loadable-modules")
endfunction()
